01  def knapsack(weights: list[int], values: list[int], W: int):
02      dp = [0] * (W + 1)
03      for w_item, val in zip(weights, values):
04          for w in range(W, w_item - 1, -1):
05              dp[w] = max(dp[w], dp[w - w_item] + val)
06      return dp[W]
